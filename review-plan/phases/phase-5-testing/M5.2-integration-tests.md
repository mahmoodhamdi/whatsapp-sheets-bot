## Test Database Setup

```typescript
// tests/integration/setup.ts
import { PrismaClient } from '@prisma/client';
import { execSync } from 'child_process';

const prisma = new PrismaClient();

export async function setupTestDatabase() {
  // Reset database
  execSync('npx prisma db push --force-reset', {
    env: { ...process.env, DATABASE_URL: process.env.TEST_DATABASE_URL },
  });

  // Seed test data
  await prisma.plan.createMany({
    data: [
      { name: 'Free', slug: 'free', messagesPerMonth: 50, rulesLimit: 1 },
      { name: 'Pro', slug: 'professional', messagesPerMonth: 5000, rulesLimit: -1 },
    ],
  });
}

export async function cleanupTestDatabase() {
  await prisma.$executeRaw`TRUNCATE TABLE users CASCADE`;
  await prisma.$disconnect();
}
```

## Integration Test Patterns

### Auth Flow Test
```typescript
// tests/integration/auth-flow.test.ts
import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { setupTestDatabase, cleanupTestDatabase } from './setup';

describe('Authentication Flow', () => {
  beforeAll(async () => {
    await setupTestDatabase();
  });

  afterAll(async () => {
    await cleanupTestDatabase();
  });

  it('should complete registration and login flow', async () => {
    // 1. Register
    const registerRes = await fetch('/api/auth/register', {
      method: 'POST',
      body: JSON.stringify({
        email: 'test@example.com',
        password: 'password123',
        name: 'Test User',
      }),
    });
    expect(registerRes.status).toBe(201);

    // 2. Verify user created
    const user = await prisma.user.findUnique({
      where: { email: 'test@example.com' },
    });
    expect(user).toBeDefined();
    expect(user?.emailVerified).toBe(false);

    // 3. Verify email
    const token = await prisma.verificationToken.findFirst({
      where: { userId: user!.id },
    });
    await fetch('/api/auth/verify-email', {
      method: 'POST',
      body: JSON.stringify({ token: token!.token }),
    });

    // 4. Login
    const loginRes = await fetch('/api/auth/callback/credentials', {
      method: 'POST',
      body: JSON.stringify({
        email: 'test@example.com',
        password: 'password123',
      }),
    });
    expect(loginRes.status).toBe(200);
  });
});
```

### Subscription Flow Test
```typescript
// tests/integration/subscription-flow.test.ts
describe('Subscription Flow', () => {
  it('should enforce free plan limits', async () => {
    // Create free user
    const user = await createTestUser('free');

    // Create rule (should work - limit is 1)
    const rule1 = await createRule(user.id, { name: 'Rule 1' });
    expect(rule1).toBeDefined();

    // Create second rule (should fail - over limit)
    const rule2Res = await fetch('/api/rules', {
      method: 'POST',
      headers: { Authorization: `Bearer ${user.token}` },
      body: JSON.stringify({ name: 'Rule 2', ... }),
    });
    expect(rule2Res.status).toBe(403);
  });

  it('should unlock features after upgrade', async () => {
    const user = await createTestUser('free');

    // Try accessing analytics (should fail)
    const analyticsRes = await fetch('/api/analytics', {
      headers: { Authorization: `Bearer ${user.token}` },
    });
    expect(analyticsRes.status).toBe(403);

    // Upgrade to professional
    await upgradeUser(user.id, 'professional');

    // Try accessing analytics (should work)
    const analyticsRes2 = await fetch('/api/analytics', {
      headers: { Authorization: `Bearer ${user.token}` },
    });
    expect(analyticsRes2.status).toBe(200);
  });
});
```

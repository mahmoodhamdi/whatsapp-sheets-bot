## Test Patterns

### Service Test
```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { prisma } from '@/lib/prisma';
import { getUserSubscription } from '@/lib/services/subscription';

vi.mock('@/lib/prisma', () => ({
  prisma: {
    subscription: {
      findUnique: vi.fn(),
    },
  },
}));

describe('getUserSubscription', () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it('should return subscription with plan', async () => {
    const mockSubscription = {
      id: '1',
      userId: 'user1',
      plan: { name: 'Professional' },
    };

    vi.mocked(prisma.subscription.findUnique).mockResolvedValue(mockSubscription);

    const result = await getUserSubscription('user1');

    expect(result).toEqual(mockSubscription);
    expect(prisma.subscription.findUnique).toHaveBeenCalledWith({
      where: { userId: 'user1' },
      include: { plan: true },
    });
  });

  it('should return null for non-existent user', async () => {
    vi.mocked(prisma.subscription.findUnique).mockResolvedValue(null);

    const result = await getUserSubscription('invalid');

    expect(result).toBeNull();
  });
});
```

### API Route Test
```typescript
import { describe, it, expect, vi } from 'vitest';
import { GET, POST } from '@/app/api/rules/route';
import { auth } from '@/lib/auth';

vi.mock('@/lib/auth');
vi.mock('@/lib/prisma');

describe('Rules API', () => {
  it('GET should return 401 without auth', async () => {
    vi.mocked(auth).mockResolvedValue(null);

    const response = await GET();

    expect(response.status).toBe(401);
  });

  it('GET should return rules for authenticated user', async () => {
    vi.mocked(auth).mockResolvedValue({ user: { id: '1' } });
    // ... mock prisma

    const response = await GET();
    const data = await response.json();

    expect(response.status).toBe(200);
    expect(data.rules).toBeDefined();
  });
});
```
